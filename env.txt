# Voice Typing Configuration

# ============================================
# MODEL SETTINGS
# ============================================
# Path to folder containing the Parakeet model (.nemo file)
# The code will automatically search for .nemo files in this folder
MODEL_FOLDER=C:\Users\LENOVO\Desktop\friday\models

# Device to use for inference
# Options: cuda (GPU) or cpu
DEVICE=cuda

# ============================================
# AUDIO SETTINGS
# ============================================
# Audio sample rate (Hz)
# Default: 16000 (recommended for Parakeet)
SAMPLE_RATE=16000

# Duration of each streaming chunk (seconds)
# Longer chunks = more accurate but slower response
# Recommended: 8-12 seconds
CHUNK_SECONDS=10

# Voice Activity Detection aggressiveness
# Range: 0 (least aggressive) to 3 (most aggressive)
# Higher = better at filtering noise, but may miss quiet speech
VAD_AGGRESSIVENESS=3

# Energy threshold multiplier for speech detection
# Higher = requires louder speech to trigger recording
# Lower = more sensitive but may trigger on background noise
ENERGY_THRESHOLD_MULTIPLIER=1.5

# ============================================
# SPEECH DETECTION
# ============================================
# Duration of silence (seconds) required to stop recording
# Increase if recording stops too early during pauses
SILENCE_DURATION=0.75

# Duration of speech (seconds) required to start recording
# Decrease for faster response, increase to avoid false triggers
SPEECH_START_DURATION=0.24

# ============================================
# UI SETTINGS
# ============================================
# Keyboard hotkey to activate/deactivate voice typing
# Format: key combinations separated by +
# Examples: ctrl+alt, ctrl+shift+v, alt+space
HOTKEY=ctrl+alt

# Size of floating indicator dot (pixels)
ICON_SIZE=40

# Opacity of floating indicator
# Range: 0.0 (invisible) to 1.0 (fully opaque)
ICON_OPACITY=0.9

# Position of floating indicator
# Options: top-right, top-left, bottom-right, bottom-left
ICON_POSITION=top-right

# ============================================
# TYPING SETTINGS
# ============================================
# Delay between typing each character (seconds)
# Lower = faster typing
# Recommended: 0.001 for instant, 0.01 for visible typing
TYPING_SPEED=0.001

# Add a space after each transcribed chunk
# Options: true or false
ADD_SPACE_AFTER_CHUNK=true
